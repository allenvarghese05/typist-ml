# T4.2 Implementation Plan — Dashboard pages (section 15.2)

## 1. Summary

This plan implements the complete visual analytical dashboard for Typist-ML specified in Technical Design §15.2. It adds backend endpoints `GET /api/v1/analysis/overview` and `GET /api/v1/analysis/participants/{pid}/trend`, hash route `#/dashboard`, and a tabbed web dashboard in `apps/web/src/features/dashboard/`. The six required views—Bigram Matrix, Per-set Outcomes, Timing, Trend, Generation, and Overhead—render strictly from real database records with zero external charting dependencies.

---

## 2. Design

- **Architecture:** Local-only FastAPI backend endpoints aggregate SQLite data into typed Pydantic models; the React 19 frontend consumes them through typed fetch clients and renders interactive tabbed views styled with Tailwind CSS and dependency-free SVG charts.
- **Routing & Deep Linking:** Extends `useHashRoute` to support `#/dashboard` with query parameters `?participant={pid}&tab={tab}` so every participant and view is bookmarkable and sharable.
- **Data Ingestion:**
  - `GET /api/v1/analysis/overview` aggregates study-wide enrolled participants, full `overhead_report` percentiles, and per-arm generation audit logs (`drills` and `generation_attempts`).
  - `GET /api/v1/analysis/participants/{pid}/trend` exposes chronological session WPM, accuracy, and practice error rates by factoring out the existing verified calculation from `typist/services/checks.py`.
  - `GET /api/v1/analysis/participants/{pid}` supplies pre-registered primary Beta posterior effects, bootstrap IKI intervals, and exclusions.
  - `GET /api/v1/analysis/sessions/{sid}/matrix` supplies the 26×26 error and IKI heatmap data.
- **Unfinished Participant Protocol (Edge Case 1):** When a participant has not completed retest (`status != 'done'`), the Outcomes & Timing view presents an informative banner ("Outcomes unlock after study completion; current phase: {phase}"), while Trend, Matrix, Generation, and Overhead views continue loading real data.
- **Zero-Dependency Charting:** Rejects third-party charting libraries (such as Recharts) to avoid React 19 / Vite 8 peer-dependency conflicts and bundle bloat, using lightweight, accessible React SVG and Tailwind CSS forest and whisker plots.

  ```python
  def participant_session_trend(session: Session, participant_id: str) -> list[SessionTrendEntry]:
      """Build session trend entries for a participant's completed sessions."""
      participant = session.get(Participant, participant_id)
      if participant is None:
          raise ParticipantNotFoundError(participant_id)

      completed_sessions = session.exec(
          select(TypingSession)
          .where(
              col(TypingSession.participant_id) == participant_id,
              col(TypingSession.status) == "completed",
          )
          .order_by(col(TypingSession.started_at).asc(), col(TypingSession.id).asc())
      ).all()

      assignments = session.exec(
          select(BigramAssignment).where(col(BigramAssignment.participant_id) == participant_id)
      ).all()
      bigram_to_set = {a.bigram: a.set_label for a in assignments}
      all_set_labels = sorted({a.set_label for a in assignments})

      trend_entries: list[SessionTrendEntry] = []
      for s in completed_sessions:
          stats = session.exec(select(BigramStat).where(col(BigramStat.session_id) == s.id)).all()
          per_set_attempts: dict[str, int] = {label: 0 for label in all_set_labels}
          per_set_errors: dict[str, int] = {label: 0 for label in all_set_labels}
          for stat in stats:
              set_label = bigram_to_set.get(stat.bigram)
              if set_label is not None:
                  per_set_attempts[set_label] += stat.attempts
                  per_set_errors[set_label] += stat.errors

          per_set_error_rate: dict[str, float | None] = {}
          for label in all_set_labels:
              att = per_set_attempts[label]
              err = per_set_errors[label]
              per_set_error_rate[label] = round(err / att, 4) if att > 0 else None

          trend_entries.append(
              SessionTrendEntry(
                  session_id=s.id,
                  phase=s.phase,
                  day_index=s.day_index,
                  started_at=s.started_at,
                  wpm=s.wpm,
                  accuracy=s.accuracy,
                  per_set_error_rate=per_set_error_rate,
                  per_set_attempts=per_set_attempts,
                  per_set_errors=per_set_errors,
              )
          )
      return trend_entries
  ```

  ```python
  def overview_report(session: Session) -> Overview:
      """Build system-wide overview: participants, overhead percentiles, and generation audit stats."""
      participants = session.exec(
          select(Participant).order_by(col(Participant.created_at).asc(), col(Participant.id).asc())
      ).all()
      participant_summaries: list[ParticipantSummary] = []
      for p in participants:
          completed_count = int(
              session.exec(
                  select(sa.func.count())
                  .select_from(TypingSession)
                  .where(
                      col(TypingSession.participant_id) == p.id,
                      col(TypingSession.status) == "completed",
                  )
              ).one()
          )
          participant_summaries.append(
              ParticipantSummary(
                  id=p.id,
                  alias=p.alias,
                  status=p.status,
                  is_builder=p.is_builder,
                  is_blinded=p.is_blinded,
                  completed_sessions=completed_count,
              )
          )

      raw_overhead = overhead_report(session, None, None)
      overhead_entries = [
          SessionOverheadEntry(
              session_id=row.session_id,
              participant_id=row.participant_id,
              events=row.events,
              handler=(
                  PercentilesSchema(p50=row.handler.p50, p95=row.handler.p95, p99=row.handler.p99)
                  if row.handler is not None
                  else None
              ),
              dispatch=(
                  PercentilesSchema(p50=row.dispatch.p50, p95=row.dispatch.p95, p99=row.dispatch.p99)
                  if row.dispatch is not None
                  else None
              ),
          )
          for row in raw_overhead
      ]

      arm_summaries: list[ArmGenerationSummary] = []
      total_drills = 0
      total_attempts = 0

      for arm in DRILL_ARMS:
          drills = session.exec(select(Drill).where(col(Drill.arm) == arm)).all()
          drills_count = len(drills)
          total_drills += drills_count

          mean_density = (
              sum(d.target_density for d in drills) / drills_count if drills_count > 0 else None
          )
          mean_word_len = (
              sum(d.mean_word_len for d in drills) / drills_count if drills_count > 0 else None
          )
          mean_real_word = (
              sum(d.real_word_ratio for d in drills) / drills_count if drills_count > 0 else None
          )

          attempts = session.exec(
              select(GenerationAttempt)
              .join(GenerationJob, col(GenerationAttempt.job_id) == col(GenerationJob.id))
              .where(col(GenerationJob.arm) == arm)
          ).all()
          attempts_count = len(attempts)
          total_attempts += attempts_count

          accepted_count = sum(1 for a in attempts if a.accepted)
          if attempts_count > 0:
              acceptance_rate = round(accepted_count / attempts_count, 4)
              attempts_per_drill = (
                  round(attempts_count / drills_count, 2) if drills_count > 0 else None
              )
          else:
              acceptance_rate = 1.0 if drills_count > 0 else None
              attempts_per_drill = 1.0 if drills_count > 0 else None

          tokens_per_sec = None
          mean_gen_ms = None
          peak_memory_mb = None

          if arm == "llm":
              tps_values = [a.tokens_per_sec for a in attempts if a.tokens_per_sec is not None]
              if tps_values:
                  tokens_per_sec = round(sum(tps_values) / len(tps_values), 1)

              gen_ms_values = [a.gen_ms for a in attempts if a.gen_ms is not None]
              if gen_ms_values:
                  mean_gen_ms = round(sum(gen_ms_values) / len(gen_ms_values), 1)

              mem_values = [a.peak_memory_mb for a in attempts if a.peak_memory_mb is not None]
              if mem_values:
                  peak_memory_mb = round(max(mem_values), 1)

          arm_summaries.append(
              ArmGenerationSummary(
                  arm=arm,
                  drills_count=drills_count,
                  attempts_count=attempts_count,
                  acceptance_rate=acceptance_rate,
                  attempts_per_drill=attempts_per_drill,
                  mean_target_density=round(mean_density, 4) if mean_density is not None else None,
                  mean_word_length=round(mean_word_len, 2) if mean_word_len is not None else None,
                  mean_real_word_ratio=round(mean_real_word, 4) if mean_real_word is not None else None,
                  tokens_per_sec=tokens_per_sec,
                  mean_gen_ms=mean_gen_ms,
                  peak_memory_mb=peak_memory_mb,
              )
          )

      return Overview(
          participants=participant_summaries,
          overhead=overhead_entries,
          generation=GenerationOverview(
              arms=arm_summaries,
              total_drills=total_drills,
              total_attempts=total_attempts,
          ),
      )
  ```

  ```typescript
          if (body.detail) msg = body.detail;
        } catch {

        return { status: "not-done", message: msg };
      }
      if (!response.ok) {
        return { status: "error", message: `HTTP ${response.status}` };
      }
      const results = (await response.json()) as ParticipantResults;
      return { status: "ok", results };
  ```

  ```typescript
  export type Route =
    | { readonly name: "demo" }
    | { readonly name: "matrix"; readonly sessionId: string }
    | { readonly name: "admin"; readonly participantId?: string }
    | { readonly name: "dashboard"; readonly participantId?: string; readonly tab?: string };

  export function parseHash(hash: string): Route {
    const matrixMatch = /^#\/matrix\/(.+)$/.exec(hash);
    const matrixEncoded = matrixMatch?.[1];
    if (matrixEncoded !== undefined) {
      return { name: "matrix", sessionId: decodeURIComponent(matrixEncoded) };
    }
    const adminMatch = /^#\/admin(?:\?(.*))?$/.exec(hash);
    if (adminMatch) {
      const search = adminMatch[1];
      const params = new URLSearchParams(search ?? "");
  ```

  ```tsx
  export interface IntervalPlotProps {
    readonly label: string;
    readonly sublabel?: string;
    readonly estimate: number | null;
    readonly lower: number | null;
    readonly upper: number | null;
    readonly minRange?: number;
    readonly maxRange?: number;
    readonly zeroLine?: boolean;
    readonly unit?: string;
    readonly higherIsBetter?: boolean;
  }

  /**
   * Pure SVG horizontal whisker / forest plot:
   * Displays point estimate (circle) with interval whiskers [lower, upper] and zero reference line.
   */
  export function IntervalPlot({
    label,
    sublabel,
    estimate,
    lower,
    upper,
    minRange = -0.15,
    maxRange = 0.15,
    zeroLine = true,
    unit = "",
    higherIsBetter = false,
  }: IntervalPlotProps): React.JSX.Element {
    if (estimate === null || lower === null || upper === null) {
      return (
        <div className="flex items-center justify-between py-1 text-sm text-gray-400">
          <span className="font-medium text-gray-700">{label}</span>
          <span>N/A</span>
        </div>
      );
    }

    const width = 280;
    const height = 36;
    const paddingX = 16;
    const scale = (val: number) => {
      const clamped = Math.max(minRange, Math.min(maxRange, val));
      const ratio = (clamped - minRange) / (maxRange - minRange);
      return paddingX + ratio * (width - 2 * paddingX);
    };

    const xZero = scale(0);
    const xEstimate = scale(estimate);
    const xLower = scale(lower);
    const xUpper = scale(upper);

    // If higherIsBetter is false (error rate & IKI: negative means improvement):
    const isImproved = higherIsBetter ? estimate > 0 : estimate < 0;
    const colorClass = isImproved ? "text-emerald-600 stroke-emerald-600 fill-emerald-600" : "text-amber-600 stroke-amber-600 fill-amber-600";

    return (
      <div className="flex flex-col sm:flex-row sm:items-center justify-between py-2 border-b border-gray-100 gap-2">
        <div className="w-44 shrink-0">
          <div className="font-medium text-sm text-gray-900">{label}</div>
          {sublabel && <div className="text-xs text-gray-500">{sublabel}</div>}
        </div>
        <div className="flex items-center gap-3">
          <svg
            width={width}
            height={height}
            role="img"
            aria-label={`${label} effect: ${estimate.toFixed(3)}${unit} with 95% interval [${lower.toFixed(3)}, ${upper.toFixed(3)}]`}
            className="overflow-visible"
          >
            {zeroLine && (
              <line
                x1={xZero}
                y1={4}
                x2={xZero}
                y2={height - 4}
                stroke="#9ca3af"
                strokeWidth={1.5}
                strokeDasharray="3 3"
              />
            )}
            {/* Horizontal whisker line */}
            <line
              x1={xLower}
              y1={height / 2}
              x2={xUpper}
              y2={height / 2}
              stroke="currentColor"
              strokeWidth={2}
              className={colorClass}
            />
            {/* Whisker caps */}
            <line
              x1={xLower}
              y1={height / 2 - 5}
              x2={xLower}
              y2={height / 2 + 5}
              stroke="currentColor"
              strokeWidth={2}
              className={colorClass}
            />
            <line
              x1={xUpper}
              y1={height / 2 - 5}
              x2={xUpper}
              y2={height / 2 + 5}
              stroke="currentColor"
              strokeWidth={2}
              className={colorClass}
            />
            {/* Estimate point */}
            <circle
              cx={xEstimate}
              cy={height / 2}
              r={4.5}
              className={colorClass}
            />
          </svg>
          <div className="text-right text-xs font-mono w-28 shrink-0">
            <span className="font-semibold text-gray-900">{estimate > 0 ? `+${estimate.toFixed(3)}` : estimate.toFixed(3)}{unit}</span>
            <div className="text-gray-500">[{lower.toFixed(3)}, {upper.toFixed(3)}]</div>
          </div>
        </div>
      </div>
    );
  }
  ```
