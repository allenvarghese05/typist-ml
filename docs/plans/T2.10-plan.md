# Implementation Plan: T2.10 Dev-only pipeline test using scripted keystrokes

## 1. Summary

Implements the end-to-end developer pipeline test simulating a participant through the complete experiment lifecycle (enrolment, consent, baseline, assignment, pool replenishment and worker generation, Latin square practice, Form B retest, data quality checks, outcome analysis, and stats recompute idempotency) in 2-set mode (`arms_enabled = ["control", "heuristic"]`). Adds a scripted keystroke generator (`generate_scripted_keystrokes`) producing valid `KeystrokeEventIn` rows with realistic dwell times, monotonic timestamps, and D14/D25 error/backspace sequences. Exposes `GET /api/v1/analysis/participants/{pid}` for outcome verification and adds a developer CLI command `typist dev test-pipeline` (`@app.command("test-pipeline")`).

## 2. Design

- **Execution Level:** Drives the full lifecycle via FastAPI HTTP `TestClient` over REST endpoints on an isolated SQLite database (`tmp_path`) with a hermetic `WordIndex`. Rejected Playwright browser smoke test for T2.10 because running 14 sessions in a browser is slow, brittle in headless CI, and reserved for frontend smoke (§17.2).
- **M2 2-Set Mode Configuration:** Configures `arms_enabled = ["control", "heuristic"]` and `practice_days = 2` to test the complete Milestone 2 cut-line (§16) without failing on unimplemented M3 arms (Markov/LLM) while keeping CI execution under 5 seconds.
- **Scripted Keystroke Generator:** `generate_scripted_keystrokes` walks `target_text`, generating `KeystrokeEventIn` rows with seeded randomness (`random.Random(seed)`). Errors inject a wrong char (`first_attempt=True`, `outcome="error"`), followed by `Backspace` (`outcome="backspace"`, `target_index=index`, `expected_char=None`), followed by the correct char (`first_attempt=False`, `outcome="correct"`). This precisely exercises D14 context error attribution and satisfies all 5 daily data quality checks.
- **Defensive SQLite Chunking:** Chunks batch inserts in `store_events` into 500-event slices, preventing SQLite `too many SQL variables` errors when large blocks are ingested in a single batch.
- **Participant Outcome Analysis:** Implements `GET /api/v1/analysis/participants/{pid}` returning `ParticipantResults` with per-set baseline and retest metrics, satisfying design doc §14.

### Step 1: Chunking in `store_events` to support large event batches

- **File touched:** `services/api/src/typist/services/events.py`
- **Exact change:**
  In `store_events`, chunk `values` in batches of 500 before executing `sqlite_insert(KeystrokeEvent).values(chunk).on_conflict_do_nothing(...)`. This prevents SQLite's parameter limit (32,766 parameters) from failing when batches with >2,000 events (15 columns each = >30,000 variables) are inserted.

```python
def store_events(
    session: Session, block_id: str, events: list[KeystrokeEventIn]
) -> EventIngestResult:
    """Insert a batch of keystroke rows for one block, idempotently, and commit."""
    if session.get(SessionBlock, block_id) is None:
        raise BlockNotFoundError(block_id)
    if not events:
        return EventIngestResult(accepted=0, duplicates=0)
    values = [{"block_id": block_id, **event.model_dump()} for event in events]
    chunk_size = 500
    accepted = 0
    for i in range(0, len(values), chunk_size):
        chunk = values[i : i + chunk_size]
        statement = (
            sqlite_insert(KeystrokeEvent)
            .values(chunk)
            .on_conflict_do_nothing(index_elements=["block_id", "seq"])
        )
        result = session.exec(statement)
        accepted += result.rowcount
    session.commit()
    return EventIngestResult(accepted=accepted, duplicates=len(events) - accepted)
```

```python
from statistics import fmean
from typist.models import BigramAssignment, Participant
from typist.services.participants import ParticipantNotFoundError
from typist.services.test_composer import ParticipantNotAssignedError


class ParticipantNotDoneError(RuntimeError):
    """The participant exists but has not completed the study (status is not 'done')."""


def participant_results(session: Session, participant_id: str) -> ParticipantResults:
    """Build ParticipantResults for a participant who has completed the study.

    Raises ParticipantNotFoundError if the participant does not exist,
    ParticipantNotDoneError if participant status is not 'done',
    and ParticipantNotAssignedError if there are no frozen bigram assignments.
    Reads bigram_stats across baseline and retest sessions.
    """
    participant = session.get(Participant, participant_id)
    if participant is None:
        raise ParticipantNotFoundError(participant_id)
    if participant.status != "done":
        raise ParticipantNotDoneError(
            f"participant {participant_id!r} status is {participant.status!r}, expected 'done'"
        )

    assignments = session.exec(
        select(BigramAssignment)
        .where(col(BigramAssignment.participant_id) == participant_id)
        .order_by(col(BigramAssignment.set_label), col(BigramAssignment.baseline_rank))
    ).all()
    if not assignments:
        raise ParticipantNotAssignedError(
            f"participant {participant_id!r} has no frozen assignments"
        )

    set_labels = sorted({a.set_label for a in assignments})
    bigram_to_set = {a.bigram: a.set_label for a in assignments}
    set_to_arm = {a.set_label: a.arm for a in assignments}
    set_to_bigrams: dict[str, list[str]] = {label: [] for label in set_labels}
    for a in assignments:
        set_to_bigrams[a.set_label].append(a.bigram)

    completed_sessions = session.exec(
        select(TypingSession)
        .where(
            col(TypingSession.participant_id) == participant_id,
            col(TypingSession.status) == "completed",
        )
    ).all()

    baseline_sessions = [s for s in completed_sessions if s.phase == "baseline"]
    retest_sessions = [s for s in completed_sessions if s.phase == "retest"]

    baseline_wpms = [s.wpm for s in baseline_sessions if s.wpm is not None]
    retest_wpms = [s.wpm for s in retest_sessions if s.wpm is not None]
    baseline_wpm = round(fmean(baseline_wpms), 2) if baseline_wpms else None
    retest_wpm = round(fmean(retest_wpms), 2) if retest_wpms else None

    baseline_session_ids = {s.id for s in baseline_sessions}
    retest_session_ids = {s.id for s in retest_sessions}

    baseline_stats = (
        session.exec(
            select(BigramStat).where(col(BigramStat.session_id).in_(baseline_session_ids))
        ).all()
        if baseline_session_ids
        else []
    )
    retest_stats = (
        session.exec(
            select(BigramStat).where(col(BigramStat.session_id).in_(retest_session_ids))
        ).all()
        if retest_session_ids
        else []
    )

    baseline_counts: dict[str, dict[str, int]] = {
        label: {"attempts": 0, "errors": 0} for label in set_labels
    }
    for stat in baseline_stats:
        set_label = bigram_to_set.get(stat.bigram)
        if set_label is not None:
            baseline_counts[set_label]["attempts"] += stat.attempts
            baseline_counts[set_label]["errors"] += stat.errors

    retest_counts: dict[str, dict[str, int]] = {
        label: {"attempts": 0, "errors": 0} for label in set_labels
    }
    for stat in retest_stats:
        set_label = bigram_to_set.get(stat.bigram)
        if set_label is not None:
            retest_counts[set_label]["attempts"] += stat.attempts
            retest_counts[set_label]["errors"] += stat.errors

    sets: list[SetOutcome] = []
    for label in set_labels:
        b_att = baseline_counts[label]["attempts"]
        b_err = baseline_counts[label]["errors"]
        b_rate = round(b_err / b_att, 4) if b_att > 0 else None

        r_att = retest_counts[label]["attempts"]
        r_err = retest_counts[label]["errors"]
        r_rate = round(r_err / r_att, 4) if r_att > 0 else None

        sets.append(
            SetOutcome(
                set_label=label,
                arm=set_to_arm[label],
                bigrams=sorted(set_to_bigrams[label]),
                baseline_attempts=b_att,
                baseline_errors=b_err,
                baseline_error_rate=b_rate,
                retest_attempts=r_att,
                retest_errors=r_err,
                retest_error_rate=r_rate,
            )
        )

    return ParticipantResults(
        participant_id=participant_id,
        alias=participant.alias,
        status=participant.status,
        baseline_wpm=baseline_wpm,
        retest_wpm=retest_wpm,
        sets=sets,
    )
```

"""Scripted keystroke generation and dev-only pipeline simulation (design doc 16; T2.10).

Provides generate_scripted_keystrokes to simulate realistic typing with monotonic timestamps,
dwell times, and backspace error corrections conforming to D14 attribution and D27 row invariants.
Provides run_pipeline_simulation to run the full end-to-end experiment loop.
Never used as scientific evidence (CLAUDE.md; Project Guide).
"""

import math
import random
from dataclasses import dataclass
from datetime import UTC, datetime
from pathlib import Path
from typing import Any

from fastapi.testclient import TestClient
from sqlalchemy import Engine
from sqlmodel import Session, col, select

from typist.corpus.build import build_corpus
from typist.corpus.word_index import WordIndex
from typist.models import (

    ExperimentConfig,

def generate_scripted_keystrokes(
    target_text: str,
    *,
    error_rate: float = 0.05,
    seed: int = 42,
    base_t_down_ms: float = 1000.0,
    mean_iki_ms: float = 140.0,
    mean_dwell_ms: float = 70.0,
) -> list[KeystrokeEventIn]:
    """Generate realistic KeystrokeEventIn rows typing target_text with seeded errors and backspaces.

    For every position pos:
    - If rng.random() < error_rate on first attempt:
      1. Type wrong character (outcome="error", first_attempt=True).
      2. Type Backspace (outcome="backspace", target_index=pos, expected_char=None).
      3. Type correct character (outcome="correct", first_attempt=False).
    - Otherwise:
      1. Type correct character (outcome="correct", first_attempt=True).

    Strictly satisfies D14 (first-attempt error attribution), D15 (monotonic timestamps),
    D24 (no unexpected pause flags), D25 (backspace post-move target_index), and D27 (row fields).
    Dwell times are between 50ms and 90ms; IKIs are positive (~140ms); sequence starts at 0 with no gaps.
    """
    if not target_text:
        return []

    rng = random.Random(seed)
    events: list[KeystrokeEventIn] = []

    t_down = base_t_down_ms
    prev_t_down: float | None = None
    attempted_positions: set[int] = set()

    seq = 0
    pos = 0
    text_len = len(target_text)

    while pos < text_len:
        expected = target_text[pos]
        first_attempt = pos not in attempted_positions
        attempted_positions.add(pos)

        make_error = first_attempt and (rng.random() < error_rate)

        if make_error:
            wrong_char = "a" if expected != "a" else "b"
            dwell = round(mean_dwell_ms + (rng.random() - 0.5) * 20.0, 2)
            iki = (
                round(mean_iki_ms + (rng.random() - 0.5) * 40.0, 2)
                if prev_t_down is not None
                else None
            )
            t_up = round(t_down + dwell, 2)
            code = "Space" if wrong_char == " " else f"Key{wrong_char.upper()}"
            events.append(
                KeystrokeEventIn(
                    seq=seq,
                    key=wrong_char,
                    code=code,
                    t_down_ms=round(t_down, 2),
                    t_up_ms=t_up,
                    dwell_ms=dwell,
                    iki_ms=iki,
                    target_index=pos,
                    expected_char=expected,
                    outcome="error",
                    first_attempt=True,
                    after_pause=False,
                    handler_ms=0.2,
                    dispatch_lag_ms=0.5,
                )
            )
            seq += 1
            prev_t_down = t_down
            t_down += round(mean_iki_ms + (rng.random() - 0.5) * 40.0, 2)

            dwell = round(mean_dwell_ms + (rng.random() - 0.5) * 20.0, 2)
            iki = round(mean_iki_ms + (rng.random() - 0.5) * 40.0, 2)
            t_up = round(t_down + dwell, 2)
            events.append(
                KeystrokeEventIn(
                    seq=seq,
                    key="Backspace",
                    code="Backspace",
                    t_down_ms=round(t_down, 2),
                    t_up_ms=t_up,
                    dwell_ms=dwell,
                    iki_ms=iki,
                    target_index=pos,
                    expected_char=None,
                    outcome="backspace",
                    first_attempt=None,
                    after_pause=False,
                    handler_ms=0.2,
                    dispatch_lag_ms=0.5,
                )
            )
            seq += 1
            prev_t_down = t_down
            t_down += round(mean_iki_ms + (rng.random() - 0.5) * 40.0, 2)

            dwell = round(mean_dwell_ms + (rng.random() - 0.5) * 20.0, 2)
            iki = round(mean_iki_ms + (rng.random() - 0.5) * 40.0, 2)
            t_up = round(t_down + dwell, 2)
            code = "Space" if expected == " " else f"Key{expected.upper()}"
            events.append(
                KeystrokeEventIn(
                    seq=seq,
                    key=expected,
                    code=code,
                    t_down_ms=round(t_down, 2),
                    t_up_ms=t_up,
                    dwell_ms=dwell,
                    iki_ms=iki,
                    target_index=pos,
                    expected_char=expected,
                    outcome="correct",
                    first_attempt=False,
                    after_pause=False,
                    handler_ms=0.2,
                    dispatch_lag_ms=0.5,
                )
            )
            seq += 1
            prev_t_down = t_down
            t_down += round(mean_iki_ms + (rng.random() - 0.5) * 40.0, 2)
            pos += 1
        else:
            dwell = round(mean_dwell_ms + (rng.random() - 0.5) * 20.0, 2)
            iki = (
                round(mean_iki_ms + (rng.random() - 0.5) * 40.0, 2)
                if prev_t_down is not None
                else None
            )
            t_up = round(t_down + dwell, 2)
            code = "Space" if expected == " " else f"Key{expected.upper()}"
            events.append(
                KeystrokeEventIn(
                    seq=seq,
                    key=expected,
                    code=code,
                    t_down_ms=round(t_down, 2),
                    t_up_ms=t_up,
                    dwell_ms=dwell,
                    iki_ms=iki,
                    target_index=pos,
                    expected_char=expected,
                    outcome="correct",
                    first_attempt=first_attempt,
                    after_pause=False,
                    handler_ms=0.2,
                    dispatch_lag_ms=0.5,
                )
            )
            seq += 1
            prev_t_down = t_down
            t_down += round(mean_iki_ms + (rng.random() - 0.5) * 40.0, 2)
            pos += 1

    return events


@dataclass(frozen=True)
class PipelineSimulationResult:
    """Summary metrics returned by run_pipeline_simulation."""

    participant_id: str
    alias: str
    status: str
    baseline_sessions: int
    practice_sessions: int
    retest_sessions: int
    total_keystrokes: int
    checks_passed: bool
    recomputed_sessions: int


def run_pipeline_simulation(
    client: TestClient,
    engine: Engine,
    db_path: Path,
    index: WordIndex,
    *,
    participant_alias: str = "pipeline-dev",
    practice_days: int = 2,
    error_rate: float = 0.05,
    seed: int = 42,
    log_fn: Any = None,
) -> PipelineSimulationResult:
    """Execute the complete experiment lifecycle from enrolment to done with scripted keystrokes."""
    def _log(msg: str) -> None:
        if log_fn:
            log_fn(msg)

    # 1. Enrolment
    _log("1. Enrolling participant...")
    resp = client.post("/api/v1/participants", json={"alias": participant_alias, "is_builder": False})
    assert resp.status_code == 201, resp.text
    pid = resp.json()["id"]

    # 2. Consent
    _log("2. Recording consent...")
    resp = client.post(f"/api/v1/participants/{pid}/consent")
    assert resp.status_code == 200, resp.text
    assert resp.json()["status"] == "baseline"

    # 3. Setup baseline passages Form A
    _log("3. Composing baseline passages (Form A)...")
    with Session(engine) as s:
        cfg = load_baseline_config(s)
        passages = compose_baseline(cfg, index, random.Random(seed))
        persist_baseline(s, passages, replace=True)

    total_keystrokes = 0

    # 4. Simulate baseline sessions (2 parts)
    _log("4. Typing baseline sessions...")
    for part in range(2):
        resp = client.post(f"/api/v1/participants/{pid}/sessions")
        assert resp.status_code == 200, resp.text
        session_plan = resp.json()
        sid = session_plan["session_id"]
        block = session_plan["blocks"][0]
        bid = block["block_id"]

        client.post(f"/api/v1/blocks/{bid}/start")
        events = generate_scripted_keystrokes(block["target_text"], error_rate=error_rate, seed=seed + part)
        total_keystrokes += len(events)
        for i in range(0, len(events), 500):
            client.post(f"/api/v1/blocks/{bid}/events", json={"events": [e.model_dump() for e in events[i : i + 500]]})
        client.post(f"/api/v1/blocks/{bid}/complete")
        complete_resp = client.post(f"/api/v1/sessions/{sid}/complete")
        assert complete_resp.status_code == 200, complete_resp.text

    # 5. Configure 2-set mode and assign
    _log("5. Configuring 2-set mode and assigning participant...")
    with Session(engine) as s:
        row_arms = s.get(ExperimentConfig, "arms_enabled")
        if row_arms:
            row_arms.value = ["control", "heuristic"]
            s.add(row_arms)
        row_pd = s.get(ExperimentConfig, "practice_days")
        if row_pd:
            row_pd.value = practice_days
            s.add(row_pd)
        s.commit()

    resp = client.post(f"/api/v1/admin/participants/{pid}/assign")
    assert resp.status_code == 200, resp.text
    assign_data = resp.json()
    assert assign_data["status"] == "assigned"
    assert assign_data["form_b_composed"] is True
    assert set(assign_data["mapping"].keys()) == {"A", "B"}

    # 6. Replenish drill pool and run worker
    _log("6. Replenishing drill pool and running generation worker...")
    resp = client.post("/api/v1/admin/pool/replenish", json={"participant_id": pid})
    assert resp.status_code == 200, resp.text
    while run_worker_once(engine, db_path, index) is not None:
        pass

    # 7. Simulate practice sessions across Latin square
    _log(f"7. Typing {practice_days} practice sessions across Latin square...")
    for day in range(1, practice_days + 1):
        resp = client.post(f"/api/v1/participants/{pid}/sessions")
        assert resp.status_code == 200, resp.text
        session_plan = resp.json()
        sid = session_plan["session_id"]
        blocks = session_plan["blocks"]
        assert len(blocks) == 2

        for b_idx, block in enumerate(blocks):
            bid = block["block_id"]
            client.post(f"/api/v1/blocks/{bid}/start")
            events = generate_scripted_keystrokes(
                block["target_text"], error_rate=error_rate, seed=seed + 100 * day + b_idx
            )
            total_keystrokes += len(events)
            for i in range(0, len(events), 500):
                client.post(f"/api/v1/blocks/{bid}/events", json={"events": [e.model_dump() for e in events[i : i + 500]]})
            client.post(f"/api/v1/blocks/{bid}/complete")

        complete_resp = client.post(f"/api/v1/sessions/{sid}/complete")
        assert complete_resp.status_code == 200, complete_resp.text

    # 8. Simulate retest sessions (2 parts)
    _log("8. Typing retest sessions (Form B)...")
    for part in range(2):
        resp = client.post(f"/api/v1/participants/{pid}/sessions")
        assert resp.status_code == 200, resp.text
        session_plan = resp.json()
        assert session_plan["phase"] == "retest"
        sid = session_plan["session_id"]
        block = session_plan["blocks"][0]
        bid = block["block_id"]

        client.post(f"/api/v1/blocks/{bid}/start")
        events = generate_scripted_keystrokes(block["target_text"], error_rate=error_rate, seed=seed + 200 + part)
        total_keystrokes += len(events)
        for i in range(0, len(events), 500):
            client.post(f"/api/v1/blocks/{bid}/events", json={"events": [e.model_dump() for e in events[i : i + 500]]})
        client.post(f"/api/v1/blocks/{bid}/complete")
        complete_resp = client.post(f"/api/v1/sessions/{sid}/complete")
        assert complete_resp.status_code == 200, complete_resp.text

    # 9. Verify checks
    _log("9. Running data checks...")
    resp = client.get(f"/api/v1/admin/checks/{pid}")
    assert resp.status_code == 200, resp.text
    checks_data = resp.json()
    assert checks_data["passed"] is True
    assert checks_data["status"] == "done"

    # 10. Verify analysis endpoint
    _log("10. Querying outcome analysis...")
    resp = client.get(f"/api/v1/analysis/participants/{pid}")
    assert resp.status_code == 200, resp.text
    analysis_data = resp.json()
    assert analysis_data["status"] == "done"
    assert len(analysis_data["sets"]) == 2

    # 11. Verify stats recompute
    _log("11. Verifying stats recompute idempotence...")
    with Session(engine) as s:
        recompute_res = recompute_stats(s, pid)
        assert recompute_res.recomputed == 2 + practice_days + 2
        assert recompute_res.skipped == 0

    return PipelineSimulationResult(
        participant_id=pid,
        alias=participant_alias,
        status="done",
        baseline_sessions=2,
        practice_sessions=practice_days,
        retest_sessions=2,
        total_keystrokes=total_keystrokes,
        checks_passed=checks_data["passed"],
        recomputed_sessions=recompute_res.recomputed,
    )
