# T2.8 Implementation Plan: Participant State Transitions

## 1. Summary

This plan implements the complete participant lifecycle state machine (`enrolled` $\rightarrow$ `baseline` $\rightarrow$ `assigned` $\rightarrow$ `practice` $\rightarrow$ `retest` $\rightarrow$ `done`, with `withdrawn` from any active phase). It provides the missing FastAPI routers and schemas for participants (`GET/POST /participants`, `POST /participants/{pid}/consent`, `POST /participants/{pid}/withdraw`, `DELETE /participants/{pid}`), admin assignment (`POST /admin/participants/{pid}/assign`), and block timing (`POST /blocks/{bid}/start`, `POST /blocks/{bid}/complete`). It hardens session planning and completion to enforce protocol integrity and advances participant status transactionally, returning HTTP 409 Conflict on all invalid state transitions.

## 2. Design

- **State machine lifecycle DAG (§3.2, §9.3 Figure 6):**
  - `enrolled` $\rightarrow$ `baseline`: Triggered by `POST /api/v1/participants/{pid}/consent`. Sets `consent_at = utc_now()`. Starting a session while `enrolled` returns 409 Conflict.
  - `baseline` $\rightarrow$ `assigned`: Triggered by `POST /api/v1/admin/participants/{pid}/assign`. Only permitted when participant has `status == "baseline"` and exactly `baseline_parts` completed baseline sessions. Executes `assign_participant` to freeze weak-bigram assignment rows and composes form B retest passages if form A exists.
  - `assigned` $\rightarrow$ `practice`: Triggered lazily in `plan_session` when the first practice session is requested and all enabled arms have $\ge \text{drills\_per\_block}$ ready drills. If the pool is not ready, raises `DrillPoolExhaustedError` (409) and leaves status as `assigned`.
  - `practice` $\rightarrow$ `retest`: Triggered in `complete_session` when completed practice sessions reach `config.practice_days` (10). Updates `participant.status = "retest"` transactionally with the session completion.
  - `retest` $\rightarrow$ `done`: Triggered in `complete_session` when completed retest sessions reach `config.retest_parts` (2). Updates `participant.status = "done"` transactionally with the session completion.
  - Active $\rightarrow$ `withdrawn`: Triggered by `POST /api/v1/participants/{pid}/withdraw`. Allowed from any active phase (`enrolled`, `baseline`, `assigned`, `practice`, `retest`). Once `withdrawn` or `done`, further session starts, block actions, session completions, or re-withdrawals return 409 Conflict.
- **Thin routers and schema boundaries (§17.1, D18):** Table models are never returned from the API; all responses use Pydantic schemas. Block and session plans omit internal arm mappings (`arm`, `passage_id`, `block_order`) so blinded participants cannot deduce arm allocations.
- **Atomic state transitions:** Participant status updates commit in the exact same transaction as their triggering mutations (`assign_participant`, `plan_session`, `complete_session`).
- *Rejected alternative:* Eager background polling for `assigned -> practice` transitions (rejected because lazy validation inside `plan_session` is race-free, transactional, and avoids a separate daemon check).

from datetime import datetime
from typist.db import utc_now


class ParticipantError(RuntimeError):
    """Base class for participant service errors."""


class ParticipantStatusConflictError(ParticipantError):
    """The participant is in an invalid status for this operation (maps to 409 Conflict)."""


class AliasAlreadyExistsError(ParticipantError):
    """A participant with this alias already exists (maps to 409 Conflict)."""


class ParticipantWithdrawnError(ParticipantError):
    """Operation rejected because the participant has withdrawn (maps to 409 Conflict)."""


def create_participant(
    session: Session,
    alias: str,
    *,
    is_builder: bool = False,
    is_blinded: bool | None = None,
    keyboard_label: str | None = None,
) -> Participant:
    """Create a new participant in 'enrolled' status. Raises AliasAlreadyExistsError on duplicate alias."""
    existing = session.exec(select(Participant).where(col(Participant.alias) == alias)).first()
    if existing is not None:
        raise AliasAlreadyExistsError(f"participant with alias {alias!r} already exists")
    blinded = False if is_builder else (is_blinded if is_blinded is not None else True)
    participant = Participant(
        alias=alias,
        is_builder=is_builder,
        is_blinded=blinded,
        keyboard_label=keyboard_label,
        status="enrolled",
    )
    session.add(participant)
    session.commit()
    session.refresh(participant)
    return participant


def list_participants(session: Session) -> list[Participant]:
    """Return all participants ordered by created_at DESC, id ASC."""
    return list(
        session.exec(
            select(Participant).order_by(col(Participant.created_at).desc(), col(Participant.id))
        ).all()
    )


def get_participant(session: Session, participant_id: str) -> Participant:
    """Load a participant by id; raises ParticipantNotFoundError if absent."""
    participant = session.get(Participant, participant_id)
    if participant is None:
        raise ParticipantNotFoundError(participant_id)
    return participant


def record_consent(
    session: Session,
    participant_id: str,
    *,
    keyboard_label: str | None = None,
    now: datetime | None = None,
) -> Participant:
    """Record consent, moving status from 'enrolled' to 'baseline' and setting consent_at.

    Raises ParticipantNotFoundError (404) if absent, or ParticipantStatusConflictError (409) if
    participant.status is not 'enrolled'.
    """
    participant = get_participant(session, participant_id)
    if participant.status != "enrolled":
        raise ParticipantStatusConflictError(
            f"cannot record consent for participant with status {participant.status!r}; must be 'enrolled'"
        )
    participant.status = "baseline"
    participant.consent_at = now if now is not None else utc_now()
    if keyboard_label is not None:
        participant.keyboard_label = keyboard_label
    session.add(participant)
    session.commit()
    session.refresh(participant)
    return participant


def withdraw_participant(session: Session, participant_id: str) -> Participant:
    """Mark an active participant 'withdrawn'.

    Raises ParticipantNotFoundError (404) if absent, or ParticipantStatusConflictError (409) if
    participant.status is already 'done' or 'withdrawn'.
    """
    participant = get_participant(session, participant_id)
    if participant.status in ("done", "withdrawn"):
        raise ParticipantStatusConflictError(
            f"cannot withdraw participant with status {participant.status!r}; must be active"
        )
    participant.status = "withdrawn"
    session.add(participant)
    session.commit()
    session.refresh(participant)
    return participant
```

"""Block timing and lifecycle service (design doc 14; T2.8)."""

from datetime import datetime

from sqlmodel import Session

from typist.db import utc_now
from typist.models import Participant, SessionBlock
from typist.models import Session as TypingSession
from typist.services.participants import ParticipantWithdrawnError


class BlockError(RuntimeError):
    """Base class for block lifecycle errors."""


class BlockNotFoundError(LookupError):
    """No session_blocks row has the given id."""


class BlockStateConflictError(BlockError):
    """Invalid block state transition (maps to 409 Conflict)."""


def start_block(session: Session, block_id: str, *, now: datetime | None = None) -> SessionBlock:
    """Set block.started_at to now.

    Raises BlockNotFoundError (404) if block does not exist. Raises BlockStateConflictError (409) if
    the session is not in_progress or if the block has already started. Raises ParticipantWithdrawnError
    (409) if the participant has withdrawn.
    """
    block = session.get(SessionBlock, block_id)
    if block is None:
        raise BlockNotFoundError(block_id)
    typing_session = session.get(TypingSession, block.session_id)
    if typing_session is None or typing_session.status != "in_progress":
        raise BlockStateConflictError("session is not in progress")
    participant = session.get(Participant, typing_session.participant_id)
    if participant is not None and participant.status == "withdrawn":
        raise ParticipantWithdrawnError(f"participant {typing_session.participant_id} is withdrawn")
    if block.started_at is not None:
        raise BlockStateConflictError("block already started")
    block.started_at = now if now is not None else utc_now()
    session.add(block)
    session.commit()
    session.refresh(block)
    return block


def complete_block(
    session: Session, block_id: str, *, now: datetime | None = None
) -> SessionBlock:
    """Set block.ended_at to now.

    Raises BlockNotFoundError (404) if block does not exist. Raises BlockStateConflictError (409) if
    the session is not in_progress, if the block has not started, or if the block has already completed.
    Raises ParticipantWithdrawnError (409) if the participant has withdrawn.
    """
    block = session.get(SessionBlock, block_id)
    if block is None:
        raise BlockNotFoundError(block_id)
    typing_session = session.get(TypingSession, block.session_id)
    if typing_session is None or typing_session.status != "in_progress":
        raise BlockStateConflictError("session is not in progress")
    participant = session.get(Participant, typing_session.participant_id)
    if participant is not None and participant.status == "withdrawn":
        raise ParticipantWithdrawnError(f"participant {typing_session.participant_id} is withdrawn")
    if block.started_at is None:
        raise BlockStateConflictError("block has not started")
    if block.ended_at is not None:
        raise BlockStateConflictError("block already completed")
    block.ended_at = now if now is not None else utc_now()
    session.add(block)
    session.commit()
    session.refresh(block)
    return block
```

"""Block timing endpoints: mark block start and end (design doc 14; T2.8)."""

from typing import Annotated

from fastapi import APIRouter, Depends, HTTPException, status
from sqlmodel import Session

from typist.db import get_session
from typist.schemas.blocks import BlockResponse
from typist.services.blocks import (
    BlockNotFoundError,
    BlockStateConflictError,
    complete_block,
    start_block,
)
from typist.services.participants import ParticipantWithdrawnError
from typist.services.session_planner import block_label

router = APIRouter(tags=["blocks"])


@router.post(
    "/blocks/{block_id}/start",
    response_model=BlockResponse,
    responses={
        status.HTTP_404_NOT_FOUND: {"description": "block not found"},
        status.HTTP_409_CONFLICT: {
            "description": "block already started, session not in progress, or participant withdrawn"
        },
    },
)
def post_block_start(
    block_id: str,
    session: Annotated[Session, Depends(get_session)],
) -> BlockResponse:
    """Record block start time; returns blinded BlockResponse."""
    try:
        block = start_block(session, block_id)
    except BlockNotFoundError as error:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND, detail="block not found"
        ) from error
    except (BlockStateConflictError, ParticipantWithdrawnError) as error:
        raise HTTPException(
            status_code=status.HTTP_409_CONFLICT, detail=str(error)
        ) from error
    return BlockResponse(
        id=block.id,
        session_id=block.session_id,
        block_index=block.block_index,
        label=block_label(block.block_index),
        started_at=block.started_at,
        ended_at=block.ended_at,
    )


@router.post(
    "/blocks/{block_id}/complete",
    response_model=BlockResponse,
    responses={
        status.HTTP_404_NOT_FOUND: {"description": "block not found"},
        status.HTTP_409_CONFLICT: {
            "description": "block not started, already completed, session not in progress, or participant withdrawn"
        },
    },
)
def post_block_complete(
    block_id: str,
    session: Annotated[Session, Depends(get_session)],
) -> BlockResponse:
    """Record block completion time; returns blinded BlockResponse."""
    try:
        block = complete_block(session, block_id)
    except BlockNotFoundError as error:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND, detail="block not found"
        ) from error
    except (BlockStateConflictError, ParticipantWithdrawnError) as error:
        raise HTTPException(
            status_code=status.HTTP_409_CONFLICT, detail=str(error)
        ) from error
    return BlockResponse(
        id=block.id,
        session_id=block.session_id,
        block_index=block.block_index,
        label=block_label(block.block_index),
        started_at=block.started_at,
        ended_at=block.ended_at,
    )

import sqlalchemy as sa
from typist.models import ExperimentConfig, Participant
from typist.services.participants import ParticipantWithdrawnError


def complete_session(session: Session, session_id: str, now: datetime) -> SessionStats:
    """Complete an in-progress session: compute stats, update participant status, write atomically.

    Raises SessionNotFoundError (404) if session does not exist, SessionNotInProgressError (409) if
    not in_progress, or ParticipantWithdrawnError (409) if participant has withdrawn.
    Advances participant status from 'practice' -> 'retest' at practice_days, and 'retest' -> 'done'
    at retest_parts (owner answer 7).
    """
    row = session.get(TypingSession, session_id)
    if row is None:
        raise SessionNotFoundError(session_id)
    if row.status != "in_progress":
        raise SessionNotInProgressError(session_id)
    participant = session.get(Participant, row.participant_id)
    if participant is not None and participant.status == "withdrawn":
        raise ParticipantWithdrawnError(f"participant {row.participant_id} is withdrawn")
    stats = compute_session_stats(session, session_id)
    row.status = "completed"
    row.ended_at = now
    session.add(row)
    session.flush()

    if participant is not None:
        completed_count = int(
            session.exec(
                select(sa.func.count())
                .select_from(TypingSession)
                .where(
                    col(TypingSession.participant_id) == row.participant_id,
                    col(TypingSession.phase) == row.phase,
                    col(TypingSession.status) == "completed",
                )
            ).one()
        )
        if row.phase == "practice":
            practice_days_row = session.get(ExperimentConfig, "practice_days")
            practice_days = int(practice_days_row.value) if practice_days_row is not None else 10
            if completed_count >= practice_days and participant.status == "practice":
                participant.status = "retest"
                session.add(participant)
        elif row.phase == "retest":
            retest_parts_row = session.get(ExperimentConfig, "retest_parts")
            retest_parts = int(retest_parts_row.value) if retest_parts_row is not None else 2
            if completed_count >= retest_parts and participant.status == "retest":
                participant.status = "done"
                session.add(participant)

    write_session_stats(session, session_id, stats)
    return stats

from typist.services.participants import (
    AliasAlreadyExistsError,
    create_participant,
    delete_participant,
)

participant_app = typer.Typer(
    help="Participant administration commands (design doc 14.1).", no_args_is_help=True
)
app.add_typer(participant_app, name="participant")


@participant_app.callback()
def participant() -> None:
    """Participant administration commands (design doc section 14.1)."""


@participant_app.command("add")
def participant_add_command(
    alias: str = typer.Option(..., "--alias", help="Unique participant alias."),
    builder: bool = typer.Option(False, "--builder", help="Flag participant as unblinded builder."),
) -> None:
    """Create a new participant in 'enrolled' status."""
    settings = get_settings()
    engine = create_db_engine(settings.db_path)
    with Session(engine) as session:
        try:
            p = create_participant(session, alias=alias, is_builder=builder)
        except AliasAlreadyExistsError:
            typer.echo(f"participant with alias {alias!r} already exists", err=True)
            raise typer.Exit(code=1) from None
    typer.echo(f"created participant {p.id} (alias={p.alias}, builder={p.is_builder})")


@participant_app.command("delete")
def participant_delete_command(
    pid: str = typer.Option(..., "--pid", help="Participant id to delete."),
) -> None:
    """Delete a participant and all child records."""
    settings = get_settings()
    engine = create_db_engine(settings.db_path)
    with Session(engine) as session:
        try:
            deletion = delete_participant(session, pid)
        except ParticipantNotFoundError:
            typer.echo(f"no participant with id {pid!r}", err=True)
            raise typer.Exit(code=1) from None
    typer.echo(
        f"deleted participant {pid}: {deletion.sessions} sessions, {deletion.drills} drills"
    )
